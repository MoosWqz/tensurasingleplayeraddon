                             .FROZEN_DISPLAY_NAME,

                             new ArrayList<>();

                                 value.trim();

                                 cleanedValue

                             cleanedValues.add(
                                     cleanedValue
                             );
                         }
                     }
 
                     if (!cleanedValues.isEmpty()) {
                         result.put(
                                 key,
                                 cleanedValues
                         );
                     }
                 }
         );
